def best_move1(board, log, team, pieces_color):
    #promocja

    pieces1 = chess_assistance.team_positions(board, team)
    punctation = []
    #silnik
    for piece1 in pieces1:
        for move1 in moves.search(board, log, piece1, pieces_color):
            punctation.append([[piece1, move1], []])
            log1 = log + [[piece1, move1]]
            board1 = moves.make(board, log1[-1])[-1]
            pieces2 = chess_assistance.team_positions(board1, team - 1)
            #przeciwnik
            chess_assistance.write(board1, pieces_color)
            for piece2 in pieces2:
                for move2 in moves.search(board1, log1, piece2, pieces_color):
                    punctation[-1][-1].append(advantage(moves.make(board1, [piece2, move2])[-1], log1 + [[piece2, move2]], pieces_color)[team] - advantage(moves.make(board1, [piece2, move2])[-1], log1 + [[piece2, move2]], pieces_color)[team - 1])
            if team < 0:
                team += 2
            if chess_assistance.checkmate(board1, team - 1, log1, pieces_color):
                return punctation[-1][0]
            if chess_assistance.stalemate(board1, team - 1, log1, pieces_color):
                if advantage(board, log, pieces_color)[team] <= advantage(board, log, pieces_color)[team - 1]:
                    return punctation[-1][0]
                else:
                    del punctation[-1]
            else:
                punctation[-1][-1] = [min(punctation[-1][-1]), sum(punctation[-1][-1]) / len(punctation[-1][-1])]
    while len(punctation) != 1:
        if punctation[0][-1][0] > punctation[1][-1][0]:

        elif punctation[0][-1][0] < punctation[1][-1][0]:
            del punctation[0]
        else:
            if punctation[0][-1][1] > punctation[1][-1][1]:
                del punctation[1]
            else:
                del punctation[0]
    return punctation[0][0]
